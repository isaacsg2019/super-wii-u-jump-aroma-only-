
#include <SDL2/SDL.h>
#include <vpad/input.h>

int main(int argc, char **argv)
{
    if (SDL_Init(SDL_INIT_VIDEO) != 0)
        return 1;

    SDL_Window *window = SDL_CreateWindow(
        "SUPER Wii U JUMP",
        SDL_WINDOWPOS_CENTERED,
        SDL_WINDOWPOS_CENTERED,
        1280, 720, 0
    );

    if (!window) {
        SDL_Quit();
        return 1;
    }

    SDL_Renderer *renderer = SDL_CreateRenderer(
        window, -1, SDL_RENDERER_ACCELERATED
    );

    if (!renderer) {
        SDL_DestroyWindow(window);
        SDL_Quit();
        return 1;
    }

    VPADStatus pad;
    VPADReadError error;

    float x = 100.0f;
    float y = 540.0f;
    float velocityY = 0.0f;

    int running = 1;
SDL_Surface *miiSurface = SDL_LoadBMP("/vol/content/mii.bmp");
SDL_Texture *miiTexture = NULL;

if (miiSurface) {
    SDL_SetColorKey(
        miiSurface,
        SDL_TRUE,
        SDL_MapRGB(miiSurface->format, 255, 0, 255)
    );

    miiTexture = SDL_CreateTextureFromSurface(renderer, miiSurface);
    SDL_FreeSurface(miiSurface);
}
    SDL_Event event;

    while (running) {
        while (SDL_PollEvent(&event)) {
            if (event.type == SDL_QUIT)
                running = 0;
        }

        VPADRead(VPAD_CHAN_0, &pad, 1, &error);

        if (error == VPAD_READ_SUCCESS) {
            if ((pad.hold & VPAD_BUTTON_LEFT) ||
                pad.leftStick.x < -0.25f)
                x -= 5.0f;

            if ((pad.hold & VPAD_BUTTON_RIGHT) ||
                pad.leftStick.x > 0.25f)
                x += 5.0f;

            if ((pad.trigger &
                (VPAD_BUTTON_A | VPAD_BUTTON_B)) &&
                y >= 540.0f)
                velocityY = -12.0f;

            if (pad.trigger & VPAD_BUTTON_PLUS)
                running = 0;
        }

        velocityY += 0.5f;
        y += velocityY;

        if (y > 540.0f) {
            y = 540.0f;
            velocityY = 0.0f;
        }

        if (x < 0.0f) x = 0.0f;
        if (x > 1232.0f) x = 1232.0f;

        // Blue sky
        SDL_SetRenderDrawColor(renderer, 100, 180, 255, 255);
        SDL_RenderClear(renderer);

        // Green ground
        SDL_Rect ground = {0, 588, 1280, 132};
        SDL_SetRenderDrawColor(renderer, 80, 160, 80, 255);
        SDL_RenderFillRect(renderer, &ground);

        // Our first playable character!
      SDL_Rect player = {(int)x, (int)y - 48, 48, 96};

if (miiTexture) {
    SDL_RenderCopy(renderer, miiTexture, NULL, &player);
} else {
    SDL_SetRenderDrawColor(renderer, 0, 255, 80, 255);
    SDL_RenderFillRect(renderer, &player);
}

        SDL_RenderPresent(renderer);
        SDL_Delay(16);
    }
if (miiTexture)
    SDL_DestroyTexture(miiTexture);
    SDL_DestroyRenderer(renderer);
    SDL_DestroyWindow(window);
    SDL_Quit();
    return 0;
}
